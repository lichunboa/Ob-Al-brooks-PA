## 📦 常用代码片段

### 1. 配置

```python
from libs.common import (
    get_db_url,
    get_service_url,
    is_docker,
    AppConfig
)

# 获取数据库URL（自动适配环境）
db_url = get_db_url()

# 获取服务URL
api_url = get_service_url('api-service')      # http://api-service:8088
telegram_url = get_service_url('telegram-service')  # http://telegram-service:8090

# 检查是否在Docker中
if is_docker():
    print("Running in Docker")

# 使用配置类
config = AppConfig()
token = config.get_bot_token()
symbols = config.get_default_symbols()
```

### 2. 日志

```python
from libs.common import setup_logging, log_execution_time

# 初始化
logger = setup_logging(
    service_name='my-service',
    level='INFO',
    enable_file=True,
    enable_console=True,
    structured=False  # True for JSON
)

# 基本日志
logger.info("服务启动")
logger.warning("警告信息")
logger.error("错误信息")

# 带上下文的日志
logger.info_ctx(
    "数据处理完成",
    symbol="BTCUSDT",
    records=1000,
    duration_ms=150
)

# 装饰器记录执行时间
@log_execution_time(logger)
def process_data():
    pass
```

### 3. 数据库

```python
from libs.common import (
    init_database,
    fetch_one,
    fetch_all,
    execute,
    get_connection
)

# 初始化（启动时调用一次）
await init_database()

# 简单查询
result = await fetch_one(
    "SELECT * FROM candles WHERE symbol = %s ORDER BY time DESC LIMIT 1",
    ("BTCUSDT",)
)

# 查询多条
results = await fetch_all(
    "SELECT * FROM candles WHERE symbol = %s AND time > %s",
    ("BTCUSDT", "2024-01-01")
)

# 执行SQL
rows_affected = await execute(
    "UPDATE signals SET status = %s WHERE id = %s",
    ("processed", 123)
)

# 使用连接
async with get_connection() as conn:
    async with conn.cursor() as cur:
        await cur.execute("SELECT * FROM candles LIMIT 10")
        rows = await cur.fetchall()
```

### 4. 健康检查

```python
from fastapi import FastAPI
from libs.common import (
    HealthChecker,
    create_health_router,
    check_database,
    check_disk_space,
    check_memory
)

app = FastAPI()

# 创建检查器
checker = HealthChecker("my-service", version="2.0.0")

# 注册检查项
checker.register("database", lambda: check_database(get_db_url()))
checker.register("disk", lambda: check_disk_space("/"))
checker.register("memory", lambda: check_memory())

# 添加路由
app.include_router(create_health_router(checker))
```

### 5. 弹性设计

```python
from libs.common import resilient, retry, circuit_breaker, rate_limit

# 重试
@retry(max_attempts=3, base_delay=1.0)
async def fetch_data():
    return await api.get_data()

# 熔断器
@circuit_breaker(name="api", failure_threshold=5, recovery_timeout=60.0)
async def call_api():
    return await external_api.call()

# 限流
@rate_limit(max_calls=10, period=1.0)
async def limited_endpoint():
    return "OK"

# 组合：熔断 + 重试
@resilient(circuit_name="api", failure_threshold=5, max_attempts=3)
async def robust_call():
    return await api.call()
```

### 6. 输入验证

```python
from libs.common import (
    Validator,
    SymbolValidator,
    TimeframeValidator,
    StringField,
    IntegerField,
    FloatField
)

# 定义验证器
validator = Validator(
    symbol=SymbolValidator(),
    timeframe=TimeframeValidator(),
    limit=IntegerField(required=False, min_value=1, max_value=1000),
    price=FloatField(required=False, min_value=0)
)

# 验证
try:
    data = validator.validate({
        "symbol": "BTCUSDT",
        "timeframe": "5m",
        "limit": 100
    })
except ValidationErrors as e:
    errors = e.to_dict()
    # 处理错误
```

## 🐛 故障排查

### 服务无法启动

```bash
# 检查日志
docker-compose logs service-name

# 检查配置
python -c "from libs.common import AppConfig; print(AppConfig().get_bot_token())"

# 检查数据库
python -c "from libs.common import get_db_url; print(get_db_url())"
```

### 健康检查失败

```bash
# 手动检查
curl -v http://localhost:8081/health

# 检查端口
netstat -tlnp | grep 8081

# 检查进程
ps aux | grep data-service
```

### 数据库连接问题

```bash

import asyncio
from libs.common import init_database, fetch_one

async def test():
    await init_database()
    result = await fetch_one("SELECT version()")
    print(result)

asyncio.run(test())
EOF
```
