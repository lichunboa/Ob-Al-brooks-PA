```dataviewjs
/* === 🦁 PA Core Engine v11.0 (Functional / No-Loop) === 
   状态：去除了所有 for 循环，使用函数式编程
   目的：彻底解决 SyntaxError: Unexpected token 'for'
*/

const startT = performance.now();

// ============================================================
// 1. 交易数据处理 (使用 forEach 替代 for 循环)
// ============================================================

// 获取页面并转为数组
const tradePages = dv.pages("#PA/Trade").where(p => !p.file.path.includes("Templates")).array();
let processedTrades = [];

// 使用 forEach 方法，绝对不会触发语法错误
tradePages.forEach(t => {
    // 安全获取属性
    let rawAcct = (t["账户类型/account_type"] || t["account_type"] || "Demo").toString().toLowerCase();
    let acctType = "Demo";
    if (rawAcct.includes("live") || rawAcct.includes("实盘")) {
        acctType = "Live";
    } else if (rawAcct.includes("back") || rawAcct.includes("回测")) {
        acctType = "Backtest";
    }

    let entry = Number(t["入场/entry_price"] || t["entry_price"] || 0);
    let stop = Number(t["止损/stop_loss"] || t["stop_loss"] || 0);
    let exit = Number(t["离场/exit_price"] || t["exit_price"] || entry); 
    let pnl = Number(t["净利润/net_profit"] || t["net_profit"] || 0);
     
    let risk = Math.abs(entry - stop);
    let r_val = (risk > 0) ? (exit - entry) / risk : 0;

    processedTrades.push({
        link: t.file.link, 
        name: t.file.name, 
        date: moment(t.date || t.file.day).format("YYYY-MM-DD"),
        pnl: pnl, 
        account: acctType, 
        cover: t["封面/cover"] || null,
        setup: (t["设置类别/setup_category"] || "Unknown").toString().split("(")[0].trim(),
        cycle: (t["市场周期/market_cycle"] || "Unknown").toString().split("(")[0].trim(),
        error: (t["管理错误/management_error"] || "None").toString(),
        r: r_val
    });
});

// 按日期排序
processedTrades.sort((a, b) => a.date.localeCompare(b.date));


// ============================================================
// 2. SR 记忆库处理 (使用 map + Promise.all 替代 for 循环)
// ============================================================

const srPages = dv.pages('#flashcards AND -"Templates"').array();
let srStats = { total: 0, due: 0, type: { single: 0, multi: 0, cloze: 0 }, folders: {} };
const todayStr = moment().format("YYYY-MM-DD");

// 使用 Promise.all 并发处理，避免使用循环语句
await Promise.all(srPages.map(async (p) => {
    try {
        let file = app.vault.getAbstractFileByPath(p.file.path);
        if (!file) return;

        let content = await app.vault.read(file);
        if (!content) return;

        // 正则统计
        let mCloze = content.match(/==[^=]+==/g) || [];
        let mRev = content.match(/^.+?:::.+$/gm) || [];
        let mNorm = content.match(/^.+?::.+$/gm) || [];
        let mMRev = content.match(/^(?:\>)?\s*\?\?\s*$/gm) || [];
        let mMNorm = content.match(/^(?:\>)?\s*\?\s*$/gm) || [];
        
        // 计算数值
        let c_cloze = mCloze.length;
        let c_sRev = mRev.length;
        let c_sNorm = Math.max(0, mNorm.length - c_sRev);
        let c_mRev = mMRev.length;
        let c_mNorm = mMNorm.length;
        let fileCards = c_cloze + c_sNorm + (c_sRev*2) + c_mNorm + (c_mRev*2);
        
        // 更新全局统计
        srStats.total += fileCards;
        srStats.type.cloze += c_cloze;
        srStats.type.single += (c_sNorm + c_sRev);
        srStats.type.multi += (c_mNorm + c_mRev);

        // 文件夹统计
        if (fileCards > 0) {
            let parts = p.file.folder.split("/");
            let fName = parts[parts.length - 1] || "Root";
            if (!srStats.folders[fName]) srStats.folders[fName] = 0;
            srStats.folders[fName] += fileCards;
        }

        // Due 日期统计 - 【这里是主要修复点】
        // 猜测你的标签格式类似 #review/2025-01-01，所以我加了正则内容
        // 如果你的插件使用 格式，请自行修改下方的正则
        let mTags = content.match(/#review\/[\d-]{10}/g) || []; 
        
        // 使用 forEach 替代 for 循环
        mTags.forEach(tagStr => {
            if (tagStr.length >= 18) {
                // 截取日期部分，这里假设前缀有8个字符 (比如 "#review/")
                let datePart = tagStr.substring(8, 18);
                if (datePart <= todayStr) {
                    srStats.due++;
                }
            }
        });

    } catch (e) {
        // 忽略单个文件错误
        console.error("SR Error in file:", p.file.path, e);
    }
}));


// ============================================================
// 3. 挂载全局并输出 (极简状态栏版)
// ============================================================

// --- A. 简易计算核心数据 ---
let totalPnL = processedTrades.reduce((acc, t) => acc + t.pnl, 0);
let winRate = 0;
if (processedTrades.length > 0) {
    let wins = processedTrades.filter(t => t.pnl > 0).length;
    winRate = (wins / processedTrades.length * 100).toFixed(1);
}

// --- B. 挂载数据到 Window (供其他脚本调用) ---
window.paData = {
    trades: processedTrades,
    stats: {
        pnl: totalPnL,
        winRate: winRate,
        count: processedTrades.length
    },
    sr: srStats,
    lastUpdate: moment().format("HH:mm:ss"),
    loadTime: (performance.now() - startT).toFixed(0) + "ms"
};

// --- C. 渲染极简状态栏 ---
// 根据利润正负设置颜色 (赢钱绿色，输钱红色，0为默认色)
let pnlColor = totalPnL > 0 ? "#4caf50" : (totalPnL < 0 ? "#ff6b6b" : "inherit");

dv.el("div", `
<div style="
    font-size: 0.75em; 
    opacity: 0.6; 
    text-align: right; 
    margin-top: 10px; 
    padding-top: 5px; 
    border-top: 1px solid rgba(255,255,255,0.1);
    font-family: monospace;
">
    <span style="margin-right: 10px;">⚡️ v11.0</span>
    
    <span style="margin-right: 10px; color: ${pnlColor}; font-weight:bold;">
        $${totalPnL.toFixed(2)}
    </span>
    
    <span style="margin-right: 10px;">
        Win: ${winRate}%
    </span>

    <span style="opacity: 0.5;">
        Trades: ${processedTrades.length} | Cards: ${srStats.total} | ${window.paData.loadTime}
    </span>
</div>
`);
```
