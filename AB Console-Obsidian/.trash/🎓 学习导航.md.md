// 2. 逻辑处理 (Data Processing)

const pages = dv.pages("#PA/Course");
const studiedMap = new Set();
const noteLinkMap = new Map();

for (let page of pages) {
    if (page.module_id) {
        let rawIds = page.module_id;
        let idArray = Array.isArray(rawIds) ? rawIds : [rawIds];
        for (let id of idArray) {
            let idStr = id.toString();
            let key = idStr.replace(/[A-Z]/g, ""); 
            noteLinkMap.set(key, page.file.link);
            if (page.studied === true) {
                studiedMap.add(key);
            }
        }
    }
}

let nextCourse = null;
let progressCount = 0;

for (let course of syllabus) {
    let key = course.id.replace(/[A-Z]/g, "");
    if (studiedMap.has(key)) {
        progressCount++;
    } else {
        if (!nextCourse) nextCourse = course;
    }
}

const percent = Math.round((progressCount / syllabus.length) * 100);
